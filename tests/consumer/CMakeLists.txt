cmake_minimum_required(VERSION 3.25)
project(AFFrameworkConsumer LANGUAGES C)

# Reuse third-party libraries when testing the installed SDK.
if(AF_CONSUMER_FREETYPE_LIBRARY)
    add_library(Freetype::Freetype UNKNOWN IMPORTED)
    set_target_properties(Freetype::Freetype PROPERTIES IMPORTED_LOCATION "${AF_CONSUMER_FREETYPE_LIBRARY}")
endif()
if(AF_CONSUMER_HARFBUZZ_LIBRARY)
    add_library(harfbuzz::harfbuzz UNKNOWN IMPORTED)
    set_target_properties(harfbuzz::harfbuzz PROPERTIES IMPORTED_LOCATION "${AF_CONSUMER_HARFBUZZ_LIBRARY}")
endif()
find_package(AFFramework CONFIG REQUIRED)
add_executable(af_framework_consumer main.c)
target_link_libraries(af_framework_consumer PRIVATE af::framework)
if(TARGET af::vulkan)
    if(NOT EXISTS "${AFFramework_SHADER_DIRECTORY}/analytic.vert.spv")
        message(FATAL_ERROR "Installed framework shader directory is missing")
    endif()
    target_compile_definitions(af_framework_consumer PRIVATE AF_CONSUMER_HAS_VULKAN=1)
endif()
